arch=arm

vendor=freescale
board=gateway_dk
board_name=gateway_dk
baudrate=115200
stderr=serial@21c0500
stdin=serial@21c0500
stdout=serial@21c0500

kernel_addr=0x800000
kernel_blk=0x4000
kernel_size=0x1800000
kernel_blkcnt=0xC000

fdt_addr_r=0x88000000
kernel_addr_r=0x82000000
kernel_comp_addr_r=0x90000000
kernel_comp_size=0x1600000
load_addr=0xa0000000
loadaddr=0xa0000000

bootargs_hwtest=
bootargs_console=console=ttyS0,115200 earlycon=uart8250,mmio,0x21c0500 panic=10
bootcmd=run emmc || run recovery
bootdelay=5

emmc_load=ext4load mmc 0:1 ${kernel_addr_r} /boot/Image.gz && ext4load mmc 0:1 ${fdt_addr_r} /boot/mono-gateway-dk.dtb
emmc_itb=ext4load mmc 0:1 ${kernel_addr_r} /boot/kernel.itb && bootm ${kernel_addr_r}
emmc=setenv bootargs "${bootargs_console} boot_medium=emmc root=/dev/mmcblk0p1 rw rootwait rootfstype=ext4 ${bootargs_hwtest}"; run emmc_load && booti ${kernel_addr_r} - ${fdt_addr_r} || run emmc_itb

ethact=fm1-mac5
ethprime=fm1-mac5
netmask=255.255.255.0
